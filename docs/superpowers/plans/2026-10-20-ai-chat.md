# AI 对话第一阶段 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 With Windows 增加一个可运行的 OpenAI 兼容文本流式对话页，并完成配置、密钥保护、Markdown 和单会话持久化。

**Architecture:** `WithWindows.Core` 负责请求构造、HTTP/SSE 协议、上下文消息、会话存储和 DPAPI 密钥保护；`WithWindows.App` 负责 WinUI 页面、配置对话框和 Markdig 原生渲染。协议层只依赖 .NET 10 内置能力，第三方只新增 `Markdig`。

**Tech Stack:** .NET 10、WinUI 3、C# latest、xunit、`HttpClient`、`System.Net.ServerSentEvents`、`System.Text.Json`、`DataProtectionProvider`、Markdig 1.4.0。

**Spec:** `docs/superpowers/specs/2026-10-20-ai-chat-design.md`

```csharp
public static string NormalizeBaseUrl(string input)
{
    string trimmed = input.Trim();
    if (!Uri.TryCreate(trimmed, UriKind.Absolute, out Uri? uri)
        || uri.Scheme is not ("http" or "https")
        || uri.Query.Length > 0
        || uri.Fragment.Length > 0)
    {
        throw new ArgumentException("请输入有效的 HTTP 或 HTTPS 地址。", nameof(input));
    }

    return trimmed.TrimEnd('/');
}

public static Uri CreateCompletionsUri(string baseUrl)
    => new(NormalizeBaseUrl(baseUrl) + "/chat/completions", UriKind.Absolute);

public static string? Validate(ChatSettings settings)
{
    if (string.IsNullOrWhiteSpace(settings.BaseUrl))
        return "请先配置 Base URL。";
    if (string.IsNullOrWhiteSpace(settings.Model))
        return "请先配置模型。";
    try
    {
        _ = NormalizeBaseUrl(settings.BaseUrl);
        return null;
    }
    catch (ArgumentException ex)
    {
        return ex.Message;
    }
}
```

```csharp
public sealed class OpenAiCompatibleChatService(HttpClient http) : IChatCompletionService
{
    public async IAsyncEnumerable<ChatDelta> StreamAsync(
        ChatSettings settings,
        string apiKey,
        IReadOnlyList<ChatMessage> messages,
        [EnumeratorCancellation] CancellationToken cancellationToken)
    {
        using HttpRequestMessage request = ChatRequestFactory.Create(settings, apiKey, messages);
        using HttpResponseMessage response = await http.SendAsync(
            request, HttpCompletionOption.ResponseHeadersRead, cancellationToken);

        if (!response.IsSuccessStatusCode)
            throw await ChatRequestException.CreateAsync(response, cancellationToken);

        await using Stream stream =
            await response.Content.ReadAsStreamAsync(cancellationToken);

        string? mediaType = response.Content.Headers.ContentType?.MediaType;
        bool isJson = mediaType is not null
            && (mediaType.EndsWith("/json", StringComparison.OrdinalIgnoreCase)
                || mediaType.EndsWith("+json", StringComparison.OrdinalIgnoreCase));

        if (!isJson)
        {
            SseParser<string> parser = SseParser.Create(stream);
            await foreach (string data in parser.EnumerateAsync(cancellationToken))
            {
                ChatStreamEvent item = ChatStreamParser.Parse(data);
                if (item.Kind == ChatStreamEventKind.Done)
                    yield break;
                if (item.Kind == ChatStreamEventKind.Error)
                    throw new ChatRequestException(null, item.Text ?? "服务返回错误。");
                if (item.Kind == ChatStreamEventKind.Content && !string.IsNullOrEmpty(item.Text))
                    yield return new ChatDelta(item.Text);
            }

            yield break;
        }

        using JsonDocument doc = await JsonDocument.ParseAsync(
            stream, cancellationToken: cancellationToken);
        ChatStreamEvent completion = ChatStreamParser.ParseCompletion(doc.RootElement.GetRawText());
        if (completion.Kind == ChatStreamEventKind.Error)
            throw new ChatRequestException(null, completion.Text ?? "服务返回错误。");
        if (completion.Kind == ChatStreamEventKind.Content && !string.IsNullOrEmpty(completion.Text))
            yield return new ChatDelta(completion.Text);
    }
}
```

`ChatRequestException` 的接口和错误提取：

```csharp
public sealed class ChatRequestException : Exception
{
    public HttpStatusCode? StatusCode { get; }
    public string UserMessage { get; }

    public ChatRequestException(HttpStatusCode? statusCode, string userMessage)
        : base(userMessage)
    {
        StatusCode = statusCode;
        UserMessage = userMessage;
    }

    public static async Task<ChatRequestException> CreateAsync(
        HttpResponseMessage response,
        CancellationToken cancellationToken)
    {
        string body = "";
        try
        {
            await using Stream stream =
                await response.Content.ReadAsStreamAsync(cancellationToken);
            using var reader = new StreamReader(stream);
            char[] buffer = new char[4096];
            int read = await reader.ReadBlockAsync(buffer.AsMemory(), cancellationToken);
            body = new string(buffer, 0, read);
        }
        catch (Exception)
        {
        }

        string? serviceMessage = null;
        try
        {
            using JsonDocument doc = JsonDocument.Parse(body);
            if (doc.RootElement.TryGetProperty("error", out JsonElement error)
                && error.TryGetProperty("message", out JsonElement message)
                && message.ValueKind == JsonValueKind.String)
            {
                serviceMessage = message.GetString();
            }
        }
        catch (JsonException)
        {
        }

        int status = (int)response.StatusCode;
        string userMessage = status switch
        {
            401 or 403 => "认证失败，请检查 API Key。",
            404 => "接口不存在，请检查 Base URL 是否包含正确路径。",
            429 => "请求过于频繁，请稍后重试。",
            >= 500 => "AI 服务暂时不可用，请稍后重试。",
            _ => "请求失败，请检查端点配置。",
        };

        return new ChatRequestException(
            response.StatusCode,
            string.IsNullOrWhiteSpace(serviceMessage)
                ? userMessage
                : userMessage + " " + serviceMessage);
    }
}
```

渲染规则：

- 非 Markdown 模式创建一个可换行 `TextBlock`，显示原始文本。
- `HeadingBlock` 渲染为半粗体，字号按级别从 24 降到 16。
- `ParagraphBlock` 渲染为 `RichTextBlock`，逐 Inline 添加 Run。
- `EmphasisInline` 对 1 个分隔符使用斜体，对 2 个使用半粗体。
- `CodeInline` 使用 `Consolas`。
- `LinkInline` 只在 `MarkdownLinkPolicy.IsAllowed(link.Url)` 时添加可点击 `Hyperlink`。
- `ListBlock` 每项生成一行 `RichTextBlock`，有序列表使用 `1.`，无序列表使用 `•`，嵌套项按层级缩进。
- `QuoteBlock` 使用左侧主题描边和缩进。
- `FencedCodeBlock` 与 `CodeBlock` 使用系统卡片背景、`CornerRadius=6`、`Consolas`、横向自动换行。
- `ThematicBreakBlock` 使用 1 DIP 的 `CardStrokeColorDefaultBrush`。
- 原始 HTML 只作为普通文本显示。
- 不实现表格、图片、数学公式和 Mermaid。

链接点击处理器捕获 `Launcher.LaunchUriAsync` 异常并写日志。

`MarkdownMessageView.xaml.cs` 的核心结构：

```csharp
public sealed partial class MarkdownMessageView : UserControl
{
    private static readonly MarkdownPipeline Pipeline =
        new MarkdownPipelineBuilder().UseAdvancedExtensions().Build();

    public static readonly DependencyProperty TextProperty =
        DependencyProperty.Register(
            nameof(Text),
            typeof(string),
            typeof(MarkdownMessageView),
            new PropertyMetadata("", OnContentChanged));

    public static readonly DependencyProperty IsMarkdownProperty =
        DependencyProperty.Register(
            nameof(IsMarkdown),
            typeof(bool),
            typeof(MarkdownMessageView),
            new PropertyMetadata(false, OnContentChanged));

    public string Text
    {
        get => (string)GetValue(TextProperty);
        set => SetValue(TextProperty, value);
    }

    public bool IsMarkdown
    {
        get => (bool)GetValue(IsMarkdownProperty);
        set => SetValue(IsMarkdownProperty, value);
    }

    public MarkdownMessageView()
    {
        InitializeComponent();
        Render();
    }

    private static void OnContentChanged(
        DependencyObject dependencyObject,
        DependencyPropertyChangedEventArgs args)
        => ((MarkdownMessageView)dependencyObject).Render();

    private void Render()
    {
        BlocksHost.Children.Clear();
        if (!IsMarkdown)
        {
            BlocksHost.Children.Add(new TextBlock
            {
                Text = Text,
                TextWrapping = TextWrapping.Wrap,
                IsTextSelectionEnabled = true,
            });
            return;
        }

        MarkdownDocument document = Markdown.Parse(Text, Pipeline);
        foreach (Block block in document)
            AddBlock(block);
    }

    private void AddBlock(Block block, int indent = 0)
    {
        switch (block)
        {
            case HeadingBlock heading:
                RichTextBlock headingText = CreateParagraph(heading, 24 - Math.Min(8, (heading.Level - 1) * 2));
                headingText.FontWeight = FontWeights.SemiBold;
                AddWithIndent(headingText, indent);
                break;
            case ParagraphBlock paragraph:
                AddWithIndent(CreateParagraph(paragraph, 14), indent);
                break;
            case ListBlock list:
                int index = 1;
                foreach (ListItemBlock item in list)
                {
                    string prefix = list.IsOrdered ? $"{index++}. " : "• ";
                    foreach (Block child in item)
                    {
                        if (child is ParagraphBlock itemParagraph)
                        {
                            RichTextBlock line = CreateParagraph(itemParagraph, 14, prefix);
                            AddWithIndent(line, indent + 1);
                        }
                        else
                        {
                            AddBlock(child, indent + 1);
                        }
                    }
                }
                break;
            case FencedCodeBlock fenced:
                AddCodeBlock(fenced.Lines.ToString(), indent);
                break;
            case CodeBlock code:
                AddCodeBlock(code.Lines.ToString(), indent);
                break;
            case QuoteBlock quote:
                var quoteBorder = new Border
                {
                    BorderThickness = new Thickness(3, 0, 0, 0),
                    Padding = new Thickness(12, 4, 0, 4),
                };
                quoteBorder.SetResourceReference(
                    Border.BorderBrushProperty,
                    "CardStrokeColorDefaultBrush");
                var quotePanel = new StackPanel { Spacing = 6 };
                quoteBorder.Child = quotePanel;
                foreach (Block child in quote)
                {
                    if (child is ParagraphBlock quoteParagraph)
                        quotePanel.Children.Add(CreateParagraph(quoteParagraph, 14));
                }
                AddWithIndent(quoteBorder, indent);
                break;
            case ThematicBreakBlock:
                var divider = new Border { Height = 1 };
                divider.SetResourceReference(
                    Border.BackgroundProperty,
                    "CardStrokeColorDefaultBrush");
                AddWithIndent(divider, indent);
                break;
            case HtmlBlock html:
                AddWithIndent(new TextBlock
                {
                    Text = html.Lines.ToString(),
                    TextWrapping = TextWrapping.Wrap,
                }, indent);
                break;
        }
    }

    private RichTextBlock CreateParagraph(
        LeafBlock block,
        double fontSize,
        string prefix = "")
    {
        var paragraph = new Paragraph();
        if (prefix.Length > 0)
            paragraph.Inlines.Add(new Run { Text = prefix });
        AddInlines(block.Inline, paragraph.Inlines);

        return new RichTextBlock
        {
            FontSize = fontSize,
            TextWrapping = TextWrapping.Wrap,
            IsTextSelectionEnabled = true,
            Blocks = { paragraph },
        };
    }

    private void AddInlines(
        ContainerInline? source,
        InlineCollection target,
        bool bold = false,
        bool italic = false)
    {
        if (source is null)
            return;

        foreach (Inline inline in source)
        {
            switch (inline)
            {
                case LiteralInline literal:
                    target.Add(CreateRun(literal.Content.ToString(), bold, italic));
                    break;
                case EmphasisInline emphasis:
                    AddInlines(
                        emphasis,
                        target,
                        bold || emphasis.DelimiterCount >= 2,
                        italic || emphasis.DelimiterCount == 1);
                    break;
                case CodeInline code:
                    target.Add(new Run
                    {
                        Text = code.Content,
                        FontFamily = new FontFamily("Consolas"),
                    });
                    break;
                case LinkInline link when MarkdownLinkPolicy.IsAllowed(link.Url):
                    var hyperlink = new Hyperlink();
                    hyperlink.Inlines.Add(CreateRun(
                        GetInlineText(link),
                        bold,
                        italic));
                    hyperlink.Click += async (_, _) =>
                    {
                        if (Uri.TryCreate(link.Url, UriKind.Absolute, out Uri? uri))
                        {
                            try
                            {
                                await Launcher.LaunchUriAsync(uri);
                            }
                            catch (Exception ex)
                            {
                                App.Host?.Log.Error($"打开 Markdown 链接失败：{ex.GetType().Name}");
                            }
                        }
                    };
                    target.Add(hyperlink);
                    break;
                case AutolinkInline autoLink
                    when MarkdownLinkPolicy.IsAllowed(autoLink.Url):
                    var autoHyperlink = new Hyperlink();
                    autoHyperlink.Inlines.Add(CreateRun(autoLink.Url, bold, italic));
                    autoHyperlink.Click += async (_, _) =>
                    {
                        if (Uri.TryCreate(autoLink.Url, UriKind.Absolute, out Uri? uri))
                        {
                            try
                            {
                                await Launcher.LaunchUriAsync(uri);
                            }
                            catch (Exception ex)
                            {
                                App.Host?.Log.Error(
                                    $"打开 Markdown 链接失败：{ex.GetType().Name}");
                            }
                        }
                    };
                    target.Add(autoHyperlink);
                    break;
                case LineBreakInline:
                    target.Add(new LineBreak());
                    break;
                case HtmlInline html:
                    target.Add(CreateRun(html.Tag, bold, italic));
                    break;
                case ContainerInline container:
                    AddInlines(container, target, bold, italic);
                    break;
            }
        }
    }

    private static Run CreateRun(string text, bool bold, bool italic)
        => new()
        {
            Text = text,
            FontWeight = bold ? FontWeights.SemiBold : FontWeights.Normal,
            FontStyle = italic ? FontStyle.Italic : FontStyle.Normal,
        };

    private void AddCodeBlock(string text, int indent)
    {
        var codeText = new TextBlock
        {
            Text = text,
            FontFamily = new FontFamily("Consolas"),
            FontSize = 13,
            TextWrapping = TextWrapping.NoWrap,
            IsTextSelectionEnabled = true,
        };
        var scroll = new ScrollViewer
        {
            HorizontalScrollBarVisibility = ScrollBarVisibility.Auto,
            VerticalScrollBarVisibility = ScrollBarVisibility.Disabled,
            HorizontalScrollMode = ScrollMode.Enabled,
            VerticalScrollMode = ScrollMode.Disabled,
            Content = codeText,
        };
        var border = new Border
        {
            Padding = new Thickness(12),
            CornerRadius = new CornerRadius(6),
            Child = scroll,
        };
        border.SetResourceReference(
            Border.BackgroundProperty,
            "LayerFillColorDefaultBrush");
        AddWithIndent(border, indent);
    }

    private void AddWithIndent(FrameworkElement element, int indent)
    {
        element.Margin = new Thickness(indent * 16, 0, 0, 0);
        BlocksHost.Children.Add(element);
    }

    private static string GetInlineText(ContainerInline? source)
    {
        if (source is null)
            return "";

        var builder = new StringBuilder();
        foreach (Inline inline in source)
        {
            switch (inline)
            {
                case LiteralInline literal:
                    builder.Append(literal.Content);
                    break;
                case CodeInline code:
                    builder.Append(code.Content);
                    break;
                case ContainerInline container:
                    builder.Append(GetInlineText(container));
                    break;
            }
        }
        return builder.ToString();
    }
```

为 `EmphasisInline.DelimiterChar == '~'` 的删除线分支设置 `Run.TextDecorations = TextDecorations.Strikethrough`；其他强调仍走粗体或斜体分支。

```csharp
public sealed class ChatMessageViewModel : INotifyPropertyChanged
{
    public ChatMessageViewModel(ChatRole role, string content, bool isMarkdown = false)
    {
        Role = role;
        _content = content;
        _isMarkdown = isMarkdown;
    }

    public ChatRole Role { get; }
    public string CopyAutomationId =>
        Role == ChatRole.User ? "ChatCopyUserMessage" : "ChatCopyAssistantMessage";
    public Visibility UserVisibility => Role == ChatRole.User ? Visibility.Visible : Visibility.Collapsed;
    public Visibility AssistantVisibility => Role == ChatRole.Assistant ? Visibility.Visible : Visibility.Collapsed;

    private string _content;
    private bool _isMarkdown;

    public event PropertyChangedEventHandler? PropertyChanged;

    public string Content
    {
        get => _content;
        set
        {
            if (_content == value) return;
            _content = value;
            PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(nameof(Content)));
        }
    }

    public bool IsMarkdown
    {
        get => _isMarkdown;
        set
        {
            if (_isMarkdown == value) return;
            _isMarkdown = value;
            PropertyChanged?.Invoke(this, new PropertyChangedEventArgs(nameof(IsMarkdown)));
        }
    }
}
```

```csharp
private async void OnLoaded(object sender, RoutedEventArgs e)
{
    if (_initialized)
        return;
    _initialized = true;

    try
    {
        ChatSession session = _historyStore.Load();
        foreach (ChatMessage message in session.Messages)
        {
            Messages.Add(new ChatMessageViewModel(
                message.Role,
                message.Content,
                isMarkdown: message.Role == ChatRole.Assistant));
        }

        _apiKey = await _secretStore.LoadAsync() ?? "";
    }
    catch (Exception ex)
    {
        App.Host?.Log.Error($"AI 会话加载失败：{ex.GetType().Name}");
        ShowError("会话文件损坏，已从空会话开始。");
    }

    UpdateEmptyState();
    ScrollToLatest();
    ChatInput.Focus(FocusState.Programmatic);
}

private void OnUnloaded(object sender, RoutedEventArgs e)
{
    _requestCts?.Cancel();
    if (Messages.Count > 0)
        SaveSession();
}
```

- [ ] **Step 4: 实现发送和停止**

```csharp
private async Task SendAsync()
{
    string text = ChatInput.Text.Trim();
    string? configError = ChatValidation.Validate(App.Host!.Config.Chat);
    if (text.Length == 0 || _isSending || configError is not null)
    {
        ShowError(configError ?? "请输入消息。");
        return;
    }

    var userMessage = new ChatMessageViewModel(ChatRole.User, text);
    Messages.Add(userMessage);
    ChatInput.Text = "";
    ChatErrorBar.IsOpen = false;

    var assistantMessage = new ChatMessageViewModel(ChatRole.Assistant, "");
    Messages.Add(assistantMessage);
    UpdateEmptyState();
    _requestCts = new CancellationTokenSource();
    SetSending(true);

    var history = Messages
        .Where(m => m.Role is ChatRole.User or ChatRole.Assistant)
        .Select(m => new ChatMessage(m.Role, m.Content))
        .ToList();

    try
    {
        var service = new OpenAiCompatibleChatService(App.Host.HttpClient);
        await foreach (ChatDelta delta in service.StreamAsync(
            App.Host.Config.Chat,
            _apiKey,
            history,
            _requestCts.Token))
        {
            assistantMessage.Content += delta.Content;
            ScrollToLatest();
        }

        assistantMessage.IsMarkdown = true;
        SaveSession();
    }
    catch (OperationCanceledException)
    {
        assistantMessage.Content = assistantMessage.Content.Length == 0
            ? "（已停止）"
            : assistantMessage.Content + "\n\n（已停止）";
        assistantMessage.IsMarkdown = true;
        SaveSession();
    }
    catch (ChatRequestException ex)
    {
        HandlePartialAndError(assistantMessage, text, ex.UserMessage);
    }
    catch (Exception ex)
    {
        App.Host.Log.Error($"AI 对话请求失败：{ex.GetType().Name}");
        HandlePartialAndError(
            assistantMessage,
            text,
            "请求失败，请检查端点配置和网络。");
    }
    finally
    {
        _requestCts.Dispose();
        _requestCts = null;
        SetSending(false);
        ChatInput.Focus(FocusState.Programmatic);
    }
}
```

```csharp
private void HandlePartialAndError(
    ChatMessageViewModel assistant,
    string originalText,
    string error)
{
    if (string.IsNullOrEmpty(assistant.Content))
    {
        Messages.Remove(assistant);
        ChatInput.Text = originalText;
    }
    else
    {
        assistant.Content += "\n\n（回复不完整）";
        assistant.IsMarkdown = true;
        SaveSession();
    }

    ShowError(error);
}

private void SaveSession()
{
    try
    {
        _historyStore.Save(new ChatSession
        {
            UpdatedAt = DateTimeOffset.Now,
            Messages = Messages
                .Where(m => m.Role is ChatRole.User or ChatRole.Assistant)
                .Select(m => new ChatMessage(m.Role, m.Content))
                .ToList(),
        });
    }
    catch (Exception ex)
    {
        App.Host?.Log.Error($"AI 会话保存失败：{ex.GetType().Name}");
    }
}

private void ShowError(string message)
{
    ChatErrorBar.Message = message;
    ChatErrorBar.IsOpen = true;
}

private void ScrollToLatest()
{
    if (Messages.Count == 0)
        return;
    ChatMessageViewModel latest = Messages[^1];
    DispatcherQueue.TryEnqueue(() =>
        MessageList.ScrollIntoView(latest, ScrollIntoViewAlignment.Leading));
}

private void SetSending(bool sending)
{
    _isSending = sending;
    ChatSendButton.Content = sending ? "停止" : "发送";
    AutomationProperties.SetName(ChatSendButton, sending ? "停止生成" : "发送消息");
}
```

```csharp
private async void OnInputKeyDown(object sender, KeyRoutedEventArgs e)
{
    if (e.Key != VirtualKey.Enter)
        return;

    bool shift =
        (InputKeyboardSource.GetKeyStateForCurrentThread(VirtualKey.Shift)
            & CoreVirtualKeyStates.Down) == CoreVirtualKeyStates.Down;
    if (shift || _isComposing)
        return;

    e.Handled = true;
    try
    {
        await SendAsync();
    }
    catch (Exception ex)
    {
        App.Host?.Log.Error($"AI 输入发送失败：{ex.GetType().Name}");
        ShowError("发送失败，请稍后重试。");
    }
}

private async void OnConfigureClick(object sender, RoutedEventArgs e)
{
    try
    {
        ChatProviderDialogResult? result = await ChatProviderDialog.ShowAsync(
            XamlRoot,
            App.Host!.Config.Chat,
            !string.IsNullOrEmpty(_apiKey));
        if (result is null)
            return;

        App.Host.Config.Chat.BaseUrl = result.Value.BaseUrl;
        App.Host.Config.Chat.Model = result.Value.Model;
        App.Host.Config.Chat.SystemPrompt = result.Value.SystemPrompt;

        if (result.Value.ClearApiKey)
        {
            _secretStore.Clear();
            _apiKey = "";
        }
        else if (!string.IsNullOrEmpty(result.Value.NewApiKey))
        {
            await _secretStore.SaveAsync(result.Value.NewApiKey);
            _apiKey = result.Value.NewApiKey;
        }

        App.Host.SaveConfig();
        ChatErrorBar.IsOpen = false;
    }
    catch (Exception ex)
    {
        App.Host?.Log.Error($"AI 配置保存失败：{ex.GetType().Name}");
        ShowError("配置保存失败，请检查文件权限。");
    }
}

private async void OnClearClick(object sender, RoutedEventArgs e)
{
    var dialog = new ContentDialog
    {
        XamlRoot = XamlRoot,
        Title = "清空会话",

        CloseButtonText = "取消",
        DefaultButton = ContentDialogButton.Close,
    };
    if (await dialog.ShowAsync() != ContentDialogResult.Primary)
        return;

    _requestCts?.Cancel();
    Messages.Clear();
    _historyStore.Clear();

}
```
